% CV template (pandoc + lualatex). Defines the macros CV variants use:
% \cvsection, \cventry, \cvdesc, \cveducation, \cvskill.
% Name and contact come from profile.json ("sender"); a variant's own
% frontmatter can override them. `photo:` is optional (path relative to the profile folder).

\setdefaultlanguage{english}

\usepackage{tikz}
\usepackage{calc}
\usepackage{array}

\usepackage{hyperref}

\definecolor{textmain}{HTML}{1a1a1a}
\definecolor{muted}{HTML}{666666}
\color{textmain}
\hypersetup{colorlinks=true, urlcolor=muted, linkcolor=textmain, pdfauthor={$if(name)$$name$$else$$sender.name$$endif$}}

\newcommand{\cventry}[3]{%
  \noindent\begin{tabular}[t]{@{}p{\leftcol}@{}p{\rightcol}@{}}
    \textcolor{muted}{#1} & \textbf{#2}\quad #3
  \end{tabular}\par\vspace{1mm}}
\newcommand{\cvdesc}[1]{\noindent\hspace{\leftcol}\parbox[t]{\rightcol}{#1}\par\vspace{0.6mm}}
\newcommand{\cveducation}[2]{%
  \noindent\begin{tabular}[t]{@{}p{\leftcol}@{}p{\rightcol}@{}}\textcolor{muted}{#1} & #2\end{tabular}\par\vspace{0.6mm}}
\newcommand{\cvskill}[2]{%
  \noindent\begin{tabular}[t]{@{}p{\leftcol}@{}p{\rightcol}@{}}#1 & #2\end{tabular}\par\vspace{0.6mm}}
